# OBD-II adapter connection info for the CYD OBD-II Dashboard.
#
# Copy this file to the ROOT of the SD card, rename it to "obd_config.txt"
# (path must be exactly /obd_config.txt), and fill in the values below for
# your own ELM327 adapter. The dashboard reads this file once at boot; there
# is no on-device page to edit these values, so re-edit this file and
# re-insert the card (or edit it on the card directly) to change them.
#
# ALL THREE FIELDS BELOW ARE REQUIRED. There is no fallback default for any
# of them - if mac, id, or password is missing, blank, or invalid, the
# device shows a red "OBD CONFIG ERROR" screen and halts at boot until this
# file is fixed and the device is rebooted.
#
# Lines starting with "#" and blank lines are ignored.

# Bluetooth MAC address of your adapter, e.g. AA:BB:CC:DD:EE:FF (colons,
# dashes, or no separator all work). Required - the dashboard always
# connects by address, never by scanning for a device name, since it's more
# reliable. Find this in your phone's Bluetooth settings or the adapter's
# documentation/box.
mac=

# Bluetooth SPP device name your adapter advertises (used for
# identification/logging only, not for connecting). Common values: OBDII,
# OBDLink, Vgate, VEEPEAK, OBD2. Required.
id=OBDII

# Bluetooth pairing PIN/password. Most ELM327 clones use a fixed legacy PIN;
# common values are 1234 or 0000. Required.
password=1234
