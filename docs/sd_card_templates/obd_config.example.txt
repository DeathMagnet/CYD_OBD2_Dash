# OBD-II adapter connection info for the CYD OBD-II Dashboard.
#
# Copy this file to the ROOT of the SD card, rename it to "obd_config.txt"
# (path must be exactly /obd_config.txt), and fill in the values below for
# your own ELM327 adapter. The dashboard reads this file once at boot;
# pre-filling it here is still the fastest way to connect on the first boot.
#
# ALL THREE FIELDS BELOW ARE REQUIRED - but there's no need to hand-edit this
# file if you don't already know your adapter's MAC address. If mac, id, or
# password is missing, blank, or invalid, OR the values here fail to connect
# 5 times in a row, the device no longer halts - it shows an on-device
# Bluetooth pairing screen (DEVICE / PASSWORD / CONNECT buttons) that scans
# for nearby adapters and rewrites this file itself once you tap CONNECT
# successfully. See the README's "On-Device Bluetooth Pairing Screen"
# section for details. You can still hand-edit this file directly (on a
# computer, or in place) and reboot to change working credentials at any
# time, without going through the pairing screen.
#
# Lines starting with "#" and blank lines are ignored.

# Bluetooth MAC address of your adapter, e.g. AA:BB:CC:DD:EE:FF (colons,
# dashes, or no separator all work). Required - the dashboard always
# connects by address, never by scanning for a device name, since it's more
# reliable. Find this in your phone's Bluetooth settings or the adapter's
# documentation/box.
mac=

# Bluetooth SPP device name your adapter advertises (used for
# identification/logging only, not for connecting). Common values: OBDII,
# OBDLink, Vgate, VEEPEAK, OBD2. Required.
id=OBDII

# Bluetooth pairing PIN/password. Most ELM327 clones use a fixed legacy PIN;
# common values are 1234 or 0000. Required.
password=1234
